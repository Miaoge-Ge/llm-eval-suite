if(NOT UNIX)
  message(FATAL_ERROR "The native service targets Linux; use a Linux container on Windows.")
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
add_library(eval_native common.cpp model_client.cpp grader.cpp engine.cpp)
target_include_directories(eval_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(eval_native PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto PkgConfig::CURL Threads::Threads)
target_compile_options(eval_native PRIVATE -Wall -Wextra -Wpedantic)
add_executable(llm-eval-server main.cpp web.cpp)
target_link_libraries(llm-eval-server PRIVATE eval_native PkgConfig::HTTPLIB)
target_compile_options(llm-eval-server PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS llm-eval-server RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(native-unit-tests tests.cpp)
  target_link_libraries(native-unit-tests PRIVATE eval_native)
  target_compile_options(native-unit-tests PRIVATE -UNDEBUG)
  add_test(NAME native-unit-tests COMMAND native-unit-tests)
endif()
